cmake_minimum_required(VERSION 3.20)
project(ExAnimaCore LANGUAGES CXX)

option(EXANIMA_BUILD_TRAINING "Build offline behavioral cloning and the training CLI" ON)
set(EXANIMA_JSON_ARCHIVE "" CACHE FILEPATH "Local copy of the pinned nlohmann JSON archive")
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
set(json_archive https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz)
if(EXANIMA_JSON_ARCHIVE)
    set(json_archive "${EXANIMA_JSON_ARCHIVE}")
endif()
set(JSON_BuildTests OFF CACHE INTERNAL "")
FetchContent_Declare(nlohmann_json URL "${json_archive}"
    URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(nlohmann_json)

add_library(exanima_core STATIC src/model.cpp)
add_library(exanima::core ALIAS exanima_core)
target_include_directories(exanima_core PUBLIC include)
target_compile_features(exanima_core PUBLIC cxx_std_17)

add_library(exanima_inference STATIC src/load.cpp)
add_library(exanima::inference ALIAS exanima_inference)
target_link_libraries(exanima_inference PUBLIC exanima_core PRIVATE nlohmann_json::nlohmann_json)
add_executable(exanima-infer app/infer.cpp)
target_include_directories(exanima-infer PRIVATE src)
target_link_libraries(exanima-infer PRIVATE exanima_inference nlohmann_json::nlohmann_json)
if(MINGW)
    target_link_options(exanima-infer PRIVATE -municode)
endif()

if(EXANIMA_BUILD_TRAINING)
    add_library(exanima_training STATIC src/training.cpp src/save.cpp)
    add_library(exanima::training ALIAS exanima_training)
    target_link_libraries(exanima_training PUBLIC exanima_core PRIVATE nlohmann_json::nlohmann_json)
    add_executable(exanima-train app/train.cpp)
    target_include_directories(exanima-train PRIVATE src)
    target_link_libraries(exanima-train PRIVATE exanima_training nlohmann_json::nlohmann_json)
    if(MINGW)
        target_link_options(exanima-train PRIVATE -municode)
    endif()
endif()

include(CTest)
if(BUILD_TESTING AND EXANIMA_BUILD_TRAINING)
    add_executable(exanima-model-test tests/model_test.cpp)
    target_include_directories(exanima-model-test PRIVATE src)
    target_link_libraries(exanima-model-test PRIVATE exanima_training exanima_inference)
    add_test(NAME core_gradients_and_model_roundtrip COMMAND exanima-model-test)
endif()
